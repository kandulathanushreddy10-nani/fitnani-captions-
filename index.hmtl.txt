function generateFitnessSRT() {
    let inputDataString = document.getElementById("fitnessScript").value.trim();
    if (!inputDataString) {
        alert("Please write or paste your gym/fitness video script lines first template update check!");
        return;
    }

    let rawLinesArray = inputDataString.split('\n');
    let srtCompiledOutputText = "";
    let continuousSecondsCounter = 1;

    for (let currentPointerIndex = 0; currentPointerIndex < rawLinesArray.length; currentPointerIndex++) {
        let singleCleanTextLine = rawLinesArray[currentPointerIndex].trim();
        if (singleCleanTextLine === "") continue;

        let chronologicalSequenceNumber = currentPointerIndex + 1;
        let startTimestampBlock = "00:00:" + String(continuousSecondsCounter).padStart(2, '0') + ",000";
        continuousSecondsCounter += 3; 
        let endTimestampBlock = "00:00:" + String(continuousSecondsCounter).padStart(2, '0') + ",000";
        continuousSecondsCounter += 1; 

        srtCompiledOutputText += chronologicalSequenceNumber + "\n";
        srtCompiledOutputText += startTimestampBlock + " --> " + endTimestampBlock + "\n";
        srtCompiledOutputText += singleCleanTextLine + "\n\n";
    }

    let textBlobDataStream = new Blob([srtCompiledOutputText], { type: "text/plain;charset=utf-8" });
    let hiddenDownloadLinkAnchor = document.createElement("a");
    hiddenDownloadLinkAnchor.href = URL.createObjectURL(textBlobDataStream);
    hiddenDownloadLinkAnchor.download = "fitnani_video_captions.srt";
    document.body.appendChild(hiddenDownloadLinkAnchor);
    hiddenDownloadLinkAnchor.click();
    document.body.removeChild(hiddenDownloadLinkAnchor);
}
</script>
